### STEP 2: Create XP Service

Create `ServerScriptService/Services/XPService.lua`:

```lua
local XPService = {}
local DataService = require(game.ServerScriptService.Services.DataService)
local LevelConfig = require(game.ServerScriptService.Config.LevelConfig)

-- Remote events for UI updates
local XPUpdateEvent = game.ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("XPUpdate")
local LevelUpEvent = game.ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("LevelUp")

-- Reward player with XP
function XPService:AddXP(userId, amount, activityType)
	local playerData = DataService:GetPlayerData(userId)
	if not playerData then return false end
	
	-- Add XP
	playerData.CurrentXP = playerData.CurrentXP + amount
	playerData.TotalXP = playerData.TotalXP + amount
	
	-- Check for level up
	local didLevelUp = self:CheckLevelUp(userId)
	
	-- Notify client
	local player = game.Players:FindFirstChild(tostring(userId))
	if player then
		XPUpdateEvent:FireClient(player, {
			CurrentXP = playerData.CurrentXP,
			Level = playerData.Level,
			TotalXP = playerData.TotalXP,
			LeveledUp = didLevelUp
		})
	end
	
	return true, amount
end

-- Check if player should level up
function XPService:CheckLevelUp(userId)
	local playerData = DataService:GetPlayerData(userId)
	if not playerData then return false end
	
	local currentLevelXP = LevelConfig:GetXPForLevel(playerData.Level)
	local nextLevelXP = LevelConfig:GetXPForLevel(playerData.Level + 1)
	
	-- Check if player has enough XP for next level
	if playerData.TotalXP >= nextLevelXP and playerData.Level < LevelConfig.MAX_LEVEL then
		return self:LevelUp(userId)
	end
	
	return false
end

-- Level up player
function XPService:LevelUp(userId)
	local playerData = DataService:GetPlayerData(userId)
	if not playerData then return false end
	
	local oldLevel = playerData.Level
	playerData.Level = playerData.Level + 1
	
	-- Grant level rewards
	local reward = LevelConfig.LEVEL_REWARDS[playerData.Level]
	if reward then
		if reward.ZBucks then
			playerData.ZBucks = playerData.ZBucks + reward.ZBucks
		end
		
		-- Grant items if any
		if reward.Items then
			for _, item in pairs(reward.Items) do
				table.insert(playerData.Inventory, item)
			end
		end
	end
	
	-- Fire level up event to client
	local player = game.Players:FindFirstChild(tostring(userId))
	if player then
		LevelUpEvent:FireClient(player, {
			NewLevel = playerData.Level,
			OldLevel = oldLevel,
			Reward = reward
		})
	end
	
	print("✓ " .. playerData.Username .. " leveled up to " .. playerData.Level)
	
	-- Recursively check for multiple level ups
	if playerData.Level < LevelConfig.MAX_LEVEL then
		return self:CheckLevelUp(userId) or true
	end
	
	return true
end

-- Get player progress to next level
function XPService:GetLevelProgress(userId)
	local playerData = DataService:GetPlayerData(userId)
	if not playerData then return 0, 0, 0 end
	
	local currentLevelXP = LevelConfig:GetXPForLevel(playerData.Level)
	local nextLevelXP = LevelConfig:GetXPForLevel(playerData.Level + 1)
	
	local xpInCurrentLevel = playerData.TotalXP - currentLevelXP
	local xpNeededForLevel = nextLevelXP - currentLevelXP
	
	return xpInCurrentLevel, xpNeededForLevel, math.floor((xpInCurrentLevel / xpNeededForLevel) * 100)
end

-- Check if player has unlocked a feature
function XPService:IsUnlocked(userId, featureName)
	local playerData = DataService:GetPlayerData(userId)
	if not playerData then return false end
	
	local unlock = LevelConfig.UNLOCKS[featureName]
	if not unlock then return false end
	
	return playerData.Level >= unlock.Level
end

return XPService
```

### STEP 3: Create Reputation System

Create `ServerScriptService/Services/ReputationService.lua`:

```lua
local ReputationService = {}
local DataService = require(game.ServerScriptService.Services.DataService)

local ReputationConfig = {
	TYPES = {
		CITIZEN = "CitizenReputation",
		CRIMINAL = "CriminalReputation",
		HUNTER = "HunterReputation",
		POLICE = "PoliceReputation"
	},
	
	TIERS = {
		{Min = 0, Max = 100, Name = "Recruit", Rank = 1},
		{Min = 100, Max = 250, Name = "Novice", Rank = 2},
		{Min = 250, Max = 500, Name = "Veteran", Rank = 3},
		{Min = 500, Max = 1000, Name = "Elite", Rank = 4},
		{Min = 1000, Max = 9999999, Name = "Legend", Rank = 5},
	}
}

-- Add reputation points
function ReputationService:AddReputation(userId, reputationType, amount)
	local playerData = DataService:GetPlayerData(userId)
	if not playerData then return false end
	
	local repKey = ReputationConfig.TYPES[reputationType]
	if not repKey then return false end
	
	playerData[repKey] = playerData[repKey] + amount
	
	print("✓ " .. playerData.Username .. " gained " .. amount .. " " .. reputationType .. " reputation")
	
	return true
end

-- Get reputation tier
function ReputationService:GetReputationTier(reputation)
	for _, tier in pairs(ReputationConfig.TIERS) do
		if reputation >= tier.Min and reputation <= tier.Max then
			return tier.Name, tier.Rank
		end
	end
	return "Unknown", 0
end

-- Get player's reputation status
function ReputationService:GetReputationStatus(userId)
	local playerData = DataService:GetPlayerData(userId)
	if not playerData then return {} end
	
	return {
		Citizen = {
			Points = playerData.CitizenReputation,
			Tier = self:GetReputationTier(playerData.CitizenReputation)
		},
		Criminal = {
			Points = playerData.CriminalReputation,
			Tier = self:GetReputationTier(playerData.CriminalReputation)
		},
		Hunter = {
			Points = playerData.HunterReputation,
			Tier = self:GetReputationTier(playerData.HunterReputation)
		},
		Police = {
			Points = playerData.PoliceReputation,
			Tier = self:GetReputationTier(playerData.PoliceReputation)
		}
	}
end

return ReputationService
```
