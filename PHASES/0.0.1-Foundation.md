### STEP 3: Create DataService (Server-Side)

**Location:** `ServerScriptService/Services/DataService.lua`

```lua
local DataService = {}
local DataStoreService = game:GetService("DataStoreService")
local RunService = game:GetService("RunService")

-- Initialize datastores
local PlayerDataStore = DataStoreService:GetDataStore("PlayerData_v1")
local TransactionLogStore = DataStoreService:GetDataStore("TransactionLogs_v1")

-- In-memory cache for active players
DataService.Cache = {}

-- Data template
local function CreatePlayerDataTemplate(userId)
	return {
		UserId = userId,
		Username = "",
		
		-- Economy
		ZBucks = 500,
		BankBalance = 0,
		
		-- Progression
		Level = 1,
		CurrentXP = 0,
		TotalXP = 0,
		
		-- Reputation
		CitizenReputation = 0,
		CriminalReputation = 0,
		HunterReputation = 0,
		PoliceReputation = 0,
		
		-- Status
		CurrentZone = "WoodStone",
		LastLogin = os.time(),
		PlayTime = 0,
		
		-- Inventory (basic)
		Inventory = {},
		
		-- Transaction History (stores last 100 transactions)
		RecentTransactions = {},
		
		-- System
		Version = 1,
		CreatedAt = os.time()
	}
end

-- Load player data
function DataService:LoadPlayerData(userId, username)
	local success, data = pcall(function()
		return PlayerDataStore:GetAsync("Player_" .. userId)
	end)
	
	if success then
		if data then
			-- Migrate old versions if needed
			data.Username = username
			data.LastLogin = os.time()
			return data
		else
			-- New player
			local newData = CreatePlayerDataTemplate(userId)
			newData.Username = username
			self:SavePlayerData(userId, newData)
			return newData
		end
	else
		warn("Failed to load data for player " .. userId .. ": " .. tostring(data))
		-- Return temporary template (will retry on next save)
		return CreatePlayerDataTemplate(userId)
	end
end

-- Save player data
function DataService:SavePlayerData(userId, data)
	local success, err = pcall(function()
		PlayerDataStore:SetAsync("Player_" .. userId, data)
	end)
	
	if not success then
		warn("Failed to save data for player " .. userId .. ": " .. tostring(err))
		return false
	end
	
	return true
end

-- Get cached player data
function DataService:GetPlayerData(userId)
	return self.Cache[userId]
end

-- Update cached player data
function DataService:UpdatePlayerData(userId, updates)
	if not self.Cache[userId] then return false end
	
	for key, value in pairs(updates) do
		self.Cache[userId][key] = value
	end
	
	return true
end

-- Log transaction
function DataService:LogTransaction(userId, transactionId, transactionType, amount, details)
	local transactionRecord = {
		TransactionID = transactionId,
		Type = transactionType,
		Amount = amount,
		Timestamp = os.time(),
		Details = details or {}
	}
	
	-- Append to recent transactions (keep last 100)
	local playerData = self:GetPlayerData(userId)
	if playerData then
		table.insert(playerData.RecentTransactions, 1, transactionRecord)
		if #playerData.RecentTransactions > 100 then
			table.remove(playerData.RecentTransactions)
		end
	end
	
	-- Also log to transaction store
	pcall(function()
		TransactionLogStore:UpdateAsync("Transactions_" .. userId, function(old)
			local transactions = old or {}
			table.insert(transactions, transactionRecord)
			return transactions
		end)
	end)
end

-- Generate unique transaction ID
function DataService:GenerateTransactionID()
	return game:GetService("HttpService"):GenerateGUID(false)
end

return DataService
```

### STEP 4: Create PlayerDataService

**Location:** `ServerScriptService/Services/PlayerDataService.lua`

```lua
local Players = game:GetService("Players")
local DataService = require(game.ServerScriptService.Services.DataService)
local Constants = require(game.ReplicatedStorage.Modules.Constants)

local PlayerDataService = {}

-- Handle player joining
function PlayerDataService:OnPlayerAdded(player)
	-- Load data
	local playerData = DataService:LoadPlayerData(player.UserId, player.Name)
	
	-- Cache it
	DataService.Cache[player.UserId] = playerData
	
	print("✓ Loaded data for " .. player.Name .. " (ID: " .. player.UserId .. ")")
	print("  Balance: " .. playerData.ZBucks .. " Z Bucks")
	print("  Level: " .. playerData.Level)
	
	return playerData
end

-- Handle player leaving
function PlayerDataService:OnPlayerRemoving(player)
	if DataService.Cache[player.UserId] then
		-- Save data before removing from cache
		local success = DataService:SavePlayerData(player.UserId, DataService.Cache[player.UserId])
		
		if success then
			print("✓ Saved data for " .. player.Name)
		else
			warn("✗ Failed to save data for " .. player.Name)
		end
		
		DataService.Cache[player.UserId] = nil
	end
end

-- Hook into player events
Players.PlayerAdded:Connect(function(player)
	PlayerDataService:OnPlayerAdded(player)
end)

Players.PlayerRemoving:Connect(function(player)
	PlayerDataService:OnPlayerRemoving(player)
end)

return PlayerDataService
```

### STEP 5: Create TransactionService

**Location:** `ServerScriptService/Services/TransactionService.lua`

```lua
local DataService = require(game.ServerScriptService.Services.DataService)
local Constants = require(game.ReplicatedStorage.Modules.Constants)

local TransactionService = {}

-- Validate transaction before processing
function TransactionService:ValidateTransaction(userId, transactionType, amount, fromBalance)
	-- Check amount
	if amount <= 0 or amount > Constants.MAX_TRANSACTION then
		return false, "Invalid amount"
	end
	
	-- Check balance for withdrawals
	if transactionType == "Withdraw" or transactionType == "Transfer" or transactionType == "Purchase" then
		if fromBalance < amount then
			return false, "Insufficient balance"
		end
	end
	
	return true, "Valid"
end

-- Process transaction
function TransactionService:ProcessTransaction(userId, transactionType, amount, targetUserId, details)
	local playerData = DataService:GetPlayerData(userId)
	if not playerData then
		return false, "Player not found"
	end
	
	-- Validate
	local isValid, message = self:ValidateTransaction(userId, transactionType, amount, playerData.ZBucks)
	if not isValid then
		return false, message
	end
	
	-- Generate transaction ID
	local transactionId = DataService:GenerateTransactionID()
	
	-- Process based on type
	if transactionType == "Deposit" then
		playerData.ZBucks = playerData.ZBucks + amount
		playerData.BankBalance = playerData.BankBalance - amount
		
	elseif transactionType == "Withdraw" then
		playerData.ZBucks = playerData.ZBucks - amount
		playerData.BankBalance = playerData.BankBalance + amount
		
	elseif transactionType == "Transfer" then
		local targetData = DataService:GetPlayerData(targetUserId)
		if not targetData then
			return false, "Target player not found"
		end
		
		playerData.ZBucks = playerData.ZBucks - amount
		targetData.ZBucks = targetData.ZBucks + amount
		
		-- Log for target too
		DataService:LogTransaction(targetUserId, transactionId, "Receive_Transfer", amount, {
			FromPlayer = userId,
			Details = details
		})
		
	elseif transactionType == "Purchase" then
		playerData.ZBucks = playerData.ZBucks - amount
	end
	
	-- Log transaction
	DataService:LogTransaction(userId, transactionId, transactionType, amount, details)
	
	return true, transactionId
end

return TransactionService
```

### STEP 8: Create Level Config

**Location:** `ServerScriptService/Config/LevelConfig.lua`

```lua
local LevelConfig = {
	MAX_LEVEL = 100,
	STARTING_LEVEL = 1,
	
	-- XP Requirements per level (cumulative)
	XP_REQUIREMENTS = {
		[1] = 0,
		[2] = 100,
		[3] = 250,
		[4] = 450,
		[5] = 700,
		-- Formula: each level requires 50 more XP than the previous
		-- Level N = 50 * N * (N-1) / 2
	},
	
	-- Generate XP requirements dynamically
	GetXPForLevel = function(self, level)
		if level <= 1 then return 0 end
		if self.XP_REQUIREMENTS[level] then
			return self.XP_REQUIREMENTS[level]
		end
		-- Formula: cumulative XP = 50 * n * (n-1) / 2
		return 50 * level * (level - 1) / 2
	end,
	
	-- Activity rewards
	ACTIVITY_REWARDS = {
		JOB_COMPLETION = 25,
		MISSION_COMPLETION = 50,
		COMBAT_KILL = 10,
		INTERACTION = 5,
	}
}

return LevelConfig
```

```lua
local RarityConfig = {
	RARITIES = {
		NORMAL = {
			Name = "Normal",
			Weight = 6000, -- 60% drop rate
			Color = Color3.new(1, 1, 1),
			Multiplier = 1.0
		},
		GOLD = {
			Name = "Gold",
			Weight = 3000, -- 30% drop rate
			Color = Color3.new(1, 0.84, 0),
			Multiplier = 2.5
		},
		RAINBOW = {
			Name = "Rainbow",
			Weight = 900, -- 9% drop rate
			Color = Color3.new(1, 0, 1),
			Multiplier = 10.0
		},
		DARK_MATTER = {
			Name = "Dark Matter",
			Weight = 100, -- 1% drop rate
			Color = Color3.new(0, 0, 0),
			Multiplier = 50.0
		}
	},
	
	-- Get random rarity based on weights
	GetRandomRarity = function(self)
		local totalWeight = 0
		for _, rarity in pairs(self.RARITIES) do
			totalWeight = totalWeight + rarity.Weight
		end
		
		local roll = math.random(1, totalWeight)
		local current = 0
		
		for name, rarity in pairs(self.RARITIES) do
			current = current + rarity.Weight
			if roll <= current then
				return name, rarity
			end
		end
		
		return "NORMAL", self.RARITIES.NORMAL
	end
}

return RarityConfig
```
